package com.evidencevault.backend.config;

import com.evidencevault.backend.entity.Role;
import com.evidencevault.backend.entity.RoleName;
import com.evidencevault.backend.entity.User;
import com.evidencevault.backend.repository.RoleRepository;
import com.evidencevault.backend.repository.UserRepository;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.CommandLineRunner;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.crypto.password.PasswordEncoder;

import java.util.LinkedHashSet;
import java.util.Set;

@Configuration
public class AdminBootstrapInitializer {

    @Value("${app.bootstrap-admin.username:}")
    private String username;

    @Value("${app.bootstrap-admin.email:}")
    private String email;

    @Value("${app.bootstrap-admin.password:}")
    private String password;

    @Bean
    CommandLineRunner initializeAdmin(
            UserRepository userRepository,
            RoleRepository roleRepository,
            PasswordEncoder passwordEncoder) {

        return args -> {

            if (username.isBlank()
                    || email.isBlank()
                    || password.isBlank()) {

                System.out.println(
                        "Admin bootstrap skipped: credentials not configured"
                );

                return;
            }

            if (password.length() < 12) {
                throw new IllegalStateException(
                        "Bootstrap admin password must be at least 12 characters"
                );
            }

            if (userRepository.existsByUsername(username)) {
                System.out.println(
                        "Admin bootstrap skipped: username already exists"
                );

                return;
            }

            if (userRepository.existsByEmail(email)) {
                throw new IllegalStateException(
                        "Bootstrap admin email already belongs to another account"
                );
            }

            Role adminRole = roleRepository
                    .findByName(RoleName.ROLE_ADMIN)
                    .orElseGet(() -> roleRepository.save(
                            new Role(RoleName.ROLE_ADMIN)
                    ));

            User admin = new User();

            admin.setUsername(username);
            admin.setEmail(email);
            admin.setPassword(passwordEncoder.encode(password));
            admin.setEnabled(true);
            admin.setRoles(
                    new LinkedHashSet<>(Set.of(adminRole))
            );

            userRepository.save(admin);

            System.out.println(
                    "Initial administrator account created successfully"
            );
        };
    }
}

package com.evidencevault.backend.config;

import com.nimbusds.jose.jwk.source.ImmutableSecret;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.oauth2.jose.jws.MacAlgorithm;
import org.springframework.security.oauth2.jwt.JwtDecoder;
import org.springframework.security.oauth2.jwt.JwtEncoder;
import org.springframework.security.oauth2.jwt.NimbusJwtDecoder;
import org.springframework.security.oauth2.jwt.NimbusJwtEncoder;

import javax.crypto.SecretKey;
import javax.crypto.spec.SecretKeySpec;
import java.util.Base64;

@Configuration
public class JwtConfig {

    @Value("${app.jwt.secret}")
    private String jwtSecret;

    @Bean
    public SecretKey jwtSecretKey() {

        byte[] decodedKey = Base64.getDecoder().decode(jwtSecret);

        if (decodedKey.length < 32) {
            throw new IllegalStateException(
                    "JWT secret must contain at least 32 bytes"
            );
        }

        return new SecretKeySpec(decodedKey, "HmacSHA256");
    }

    @Bean
    public JwtEncoder jwtEncoder(SecretKey jwtSecretKey) {

        return new NimbusJwtEncoder(
                new ImmutableSecret<>(jwtSecretKey)
        );
    }

    @Bean
    public JwtDecoder jwtDecoder(SecretKey jwtSecretKey) {

        return NimbusJwtDecoder
                .withSecretKey(jwtSecretKey)
                .macAlgorithm(MacAlgorithm.HS256)
                .build();
    }
}

package com.evidencevault.backend.config;

import java.time.Duration;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.client.SimpleClientHttpRequestFactory;
import org.springframework.web.client.RestClient;

@Configuration
public class PythonVerifierConfig {

    @Bean
    public RestClient pythonVerifierRestClient(
            @Value("${app.python-verifier.base-url}") String baseUrl,
            @Value("${app.python-verifier.service-token}") String serviceToken) {

        if (serviceToken == null || serviceToken.isBlank()) {
            throw new IllegalStateException(
                    "Python verifier service token is missing"
            );
        }

        SimpleClientHttpRequestFactory requestFactory =
                new SimpleClientHttpRequestFactory();

        requestFactory.setConnectTimeout(Duration.ofSeconds(5));
        requestFactory.setReadTimeout(Duration.ofSeconds(90));

        return RestClient.builder()
                .baseUrl(baseUrl)
                .requestFactory(requestFactory)
                .defaultHeader("X-Service-Token", serviceToken)
                .build();
    }
}

package com.evidencevault.backend.config;

import com.evidencevault.backend.service.CustomUserDetailsService;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.authentication.AuthenticationManager;
import org.springframework.security.authentication.ProviderManager;
import org.springframework.security.authentication.dao.DaoAuthenticationProvider;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.oauth2.server.resource.authentication.JwtAuthenticationConverter;
import org.springframework.security.oauth2.server.resource.authentication.JwtGrantedAuthoritiesConverter;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
@EnableWebSecurity
@EnableMethodSecurity
public class SecurityConfig {

    private final CustomUserDetailsService userDetailsService;

    public SecurityConfig(
            CustomUserDetailsService userDetailsService) {

        this.userDetailsService = userDetailsService;
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    @Bean
    public DaoAuthenticationProvider authenticationProvider() {

        DaoAuthenticationProvider provider =
                new DaoAuthenticationProvider(userDetailsService);

        provider.setPasswordEncoder(passwordEncoder());

        return provider;
    }

    @Bean
    public AuthenticationManager authenticationManager() {

        return new ProviderManager(authenticationProvider());
    }

    @Bean
    public JwtAuthenticationConverter jwtAuthenticationConverter() {

        JwtGrantedAuthoritiesConverter authoritiesConverter =
                new JwtGrantedAuthoritiesConverter();

        authoritiesConverter.setAuthoritiesClaimName("roles");
        authoritiesConverter.setAuthorityPrefix("");

        JwtAuthenticationConverter converter =
                new JwtAuthenticationConverter();

        converter.setJwtGrantedAuthoritiesConverter(
                authoritiesConverter
        );

        return converter;
    }

    @Bean
    public SecurityFilterChain securityFilterChain(
            HttpSecurity http) throws Exception {

        http
                .csrf(csrf -> csrf.disable())

                .sessionManagement(session -> session
                        .sessionCreationPolicy(
                                SessionCreationPolicy.STATELESS
                        )
                )

                .authorizeHttpRequests(auth -> auth

                        .requestMatchers(
                                "/api/v1/auth/register",
                                "/api/v1/auth/login",
                                "/actuator/health"
                        ).permitAll()

                        .requestMatchers("/api/v1/admin/**")
                        .hasAuthority("ROLE_ADMIN")

                        .requestMatchers("/api/v1/auditor/**")
                        .hasAnyAuthority(
                                "ROLE_ADMIN",
                                "ROLE_AUDITOR"
                        )

                        .anyRequest().authenticated()
                )

                .authenticationProvider(authenticationProvider())

                .oauth2ResourceServer(oauth2 -> oauth2
                        .jwt(jwt -> jwt
                                .jwtAuthenticationConverter(
                                        jwtAuthenticationConverter()
                                )
                        )
                );

        return http.build();
    }
}

package com.evidencevault.backend.exception;

import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.security.core.AuthenticationException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import com.evidencevault.backend.exception.DuplicateDocumentException;

import java.util.LinkedHashMap;
import java.util.Map;

@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(ResourceConflictException.class)
    public ProblemDetail handleConflict(
            ResourceConflictException exception) {

        return ProblemDetail.forStatusAndDetail(
                HttpStatus.CONFLICT,
                exception.getMessage()
        );
    }

    @ExceptionHandler(AuthenticationException.class)
    public ProblemDetail handleAuthentication(
            AuthenticationException exception) {

        return ProblemDetail.forStatusAndDetail(
                HttpStatus.UNAUTHORIZED,
                "Invalid username or password"
        );
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ProblemDetail handleValidation(
            MethodArgumentNotValidException exception) {

        Map<String, String> errors = new LinkedHashMap<>();

        exception.getBindingResult()
                .getFieldErrors()
                .forEach(error ->
                        errors.putIfAbsent(
                                error.getField(),
                                error.getDefaultMessage()
                        )
                );

        ProblemDetail problem = ProblemDetail.forStatusAndDetail(
                HttpStatus.BAD_REQUEST,
                "Request validation failed"
        );

        problem.setProperty("fieldErrors", errors);

        return problem;
    }

    @ExceptionHandler(IllegalArgumentException.class)
    public ProblemDetail handleBadRequest(
            IllegalArgumentException exception) {

        return ProblemDetail.forStatusAndDetail(
                HttpStatus.BAD_REQUEST,
                exception.getMessage()
        );
    }

    @ExceptionHandler(DocumentIntegrityException.class)
    public ProblemDetail handleIntegrityFailure(
            DocumentIntegrityException exception) {

        return ProblemDetail.forStatusAndDetail(
                HttpStatus.CONFLICT,
                exception.getMessage()
        );
    }

    @ExceptionHandler(java.util.NoSuchElementException.class)
    public ProblemDetail handleNotFound(
            java.util.NoSuchElementException exception) {

        return ProblemDetail.forStatusAndDetail(
                HttpStatus.NOT_FOUND,
                exception.getMessage()
        );
    }

    @ExceptionHandler(DuplicateDocumentException.class)
    public ProblemDetail handleDuplicate(
            DuplicateDocumentException exception) {

        return ProblemDetail.forStatusAndDetail(
                HttpStatus.CONFLICT,
                exception.getMessage()
        );
    }


}

            );

                jwtService.getExpirationSeconds(),

    @Transactional
    public DocumentResponse upload(
            MultipartFile file,
            DocumentCategory category,
            String description,
            String username) throws IOException {

        validateFile(file);

        User uploader = userRepository.findByUsername(username)
                .orElseThrow(() ->
                        new IllegalArgumentException(
                                "Authenticated user was not found"
                        )
                );

        String originalFilename =
                sanitizeFilename(file.getOriginalFilename());

        String extension = getExtension(originalFilename);

        String expectedContentType = ALLOWED_TYPES.get(extension);

        String actualContentType = file.getContentType();

        if (expectedContentType == null
                || !expectedContentType.equalsIgnoreCase(
                actualContentType == null
                        ? ""
                        : actualContentType
        )) {

            throw new IllegalArgumentException(
                    "Unsupported file type or content type mismatch"
            );
        }

        String checksum = calculateChecksumAndValidate(file, extension);

        if (documentRepository.existsBySha256ChecksumAndStatus(
                checksum,
                DocumentStatus.ACTIVE)) {

            throw new DuplicateDocumentException(
                    "An active document with identical content already exists"
            );
        }

        String storageKey = UUID.randomUUID().toString();

        Document document = new Document();

        document.setOriginalFilename(originalFilename);
        document.setStorageKey(storageKey);
        document.setContentType(expectedContentType);
        document.setFileSizeBytes(file.getSize());
        document.setSha256Checksum(checksum);
        document.setCategory(category);
        document.setDescription(description);
        document.setStatus(DocumentStatus.ACTIVE);
        document.setUploadedBy(uploader);

        boolean fileStored = false;

        try (InputStream inputStream = file.getInputStream()) {

            String versionId = storageService.storeVersioned(
                    storageKey,
                    inputStream,
                    expectedContentType,
                    originalFilename
            );

            fileStored = true;

            Document savedDocument =
                    documentRepository.save(document);

            saveVersionMetadata(
                    savedDocument,
                    versionId,
                    originalFilename,
                    expectedContentType,
                    file.getSize(),
                    checksum
            );

            auditService.record(
                    username,
                    "DOCUMENT_UPLOAD",
                    savedDocument.getId(),
                    "Document uploaded: " + savedDocument.getOriginalFilename()
            );

            return toResponse(savedDocument);

        } catch (Exception exception) {

            if (fileStored) {
                try {
                    storageService.delete(storageKey);
                } catch (IOException cleanupException) {
                    exception.addSuppressed(cleanupException);
                }
            }

            if (exception instanceof IOException ioException) {
                throw ioException;
            }

            throw new IllegalStateException(
                    "Document upload could not be completed",
                    exception
            );
        }
    }

    private void validateFile(MultipartFile file) {

        if (file == null || file.isEmpty()) {
            throw new IllegalArgumentException(
                    "Please select a non-empty file"
            );
        }

        if (file.getSize() > maxFileSize) {
            throw new IllegalArgumentException(

    private String sanitizeFilename(String filename) {

        if (filename == null || filename.isBlank()) {
            throw new IllegalArgumentException(
                    "Filename is required"
            );
        }

        String safeFilename = filename
                .replace("\\", "/");

        safeFilename = safeFilename.substring(
                safeFilename.lastIndexOf('/') + 1
        );

        safeFilename = safeFilename
                .replaceAll("[\\p{Cntrl}]", "")
                .trim();

        if (safeFilename.isBlank() || safeFilename.length() > 255) {
            throw new IllegalArgumentException(
                    "Invalid filename"
            );
        }

        return safeFilename;
    }

    private String getExtension(String filename) {

        int dotIndex = filename.lastIndexOf('.');

        if (dotIndex < 0 || dotIndex == filename.length() - 1) {
            throw new IllegalArgumentException(
                    "File extension is required"
            );
        }

        return filename.substring(dotIndex + 1)
                .toLowerCase(Locale.ROOT);
    }

    private String calculateChecksumAndValidate(
            MultipartFile file,
            String extension) throws IOException {

        try {

            MessageDigest digest =
                    MessageDigest.getInstance("SHA-256");

            byte[] header;

            try (InputStream inputStream = file.getInputStream()) {

                header = inputStream.readNBytes(8);

                digest.update(header);

                byte[] buffer = new byte[8192];

                int bytesRead;

                while ((bytesRead = inputStream.read(buffer)) != -1) {
                    digest.update(buffer, 0, bytesRead);
                }
            }

            validateFileSignature(header, extension);

            return HexFormat.of().formatHex(digest.digest());

        } catch (NoSuchAlgorithmException exception) {

            throw new IllegalStateException(
                    "SHA-256 algorithm is unavailable",
                    exception
            );
        }
    }

    private void validateFileSignature(
            byte[] header,
            String extension) {

        boolean valid = switch (extension) {

            case "pdf" ->
                    startsWith(header, new byte[]{
                            0x25, 0x50, 0x44, 0x46, 0x2D
                    });

            case "png" ->
                    startsWith(header, new byte[]{
                            (byte) 0x89, 0x50, 0x4E, 0x47,
                            0x0D, 0x0A, 0x1A, 0x0A
                    });

            case "jpg", "jpeg" ->
                    startsWith(header, new byte[]{
                            (byte) 0xFF, (byte) 0xD8, (byte) 0xFF
                    });

            case "txt" ->
                    header.length > 0 && !containsNullByte(header);

            default -> false;
        };

        if (!valid) {
            throw new IllegalArgumentException(
                    "File content does not match its extension"
            );
        }
    }

    private boolean startsWith(byte[] actual, byte[] expected) {

        if (actual.length < expected.length) {
            return false;
        }

        for (int i = 0; i < expected.length; i++) {
            if (actual[i] != expected[i]) {
                return false;
            }
        }

        return true;
    }

    private boolean containsNullByte(byte[] bytes) {

        for (byte value : bytes) {
            if (value == 0) {
                return true;
            }
        }

        return false;
    }

    private DocumentDownload loadVerifiedDocument(
            Long documentId,
            String username,
            String auditAction,
            String auditDetails,
            String unavailableMessage
    ) throws IOException {

        Document document = documentRepository.findById(documentId)
                .orElseThrow(() ->
                        new java.util.NoSuchElementException(
                                "Document not found"
                        )
                );

        // Only ACTIVE documents can be viewed or downloaded.
        if (document.getStatus() != DocumentStatus.ACTIVE) {
            throw new IllegalArgumentException(unavailableMessage);
        }

        // Load the current object from the configured storage provider.
        Resource resource = storageService.load(
                document.getStorageKey()
        );

        String actualChecksum;

        try (InputStream input = resource.getInputStream()) {

            MessageDigest digest = MessageDigest.getInstance("SHA-256");

            byte[] buffer = new byte[8192];

            int bytesRead;

            while ((bytesRead = input.read(buffer)) != -1) {
                digest.update(buffer, 0, bytesRead);
            }

            actualChecksum = HexFormat.of()
                    .formatHex(digest.digest());

        } catch (NoSuchAlgorithmException exception) {

            throw new IllegalStateException(
                    "SHA-256 algorithm is unavailable",
                    exception
            );
        }

        String expectedChecksum = document.getSha256Checksum();

        boolean checksumMatches =
                expectedChecksum != null
                        && MessageDigest.isEqual(
                        expectedChecksum.getBytes(
                                StandardCharsets.US_ASCII
                        ),
                        actualChecksum.getBytes(
                                StandardCharsets.US_ASCII
                        )
                );

        // Never return a document whose integrity check fails.
        if (!checksumMatches) {

            document.setStatus(DocumentStatus.CORRUPTED);

            documentRepository.save(document);

            auditService.record(
                    username,
                    "DOCUMENT_INTEGRITY_FAILURE",
                    documentId,
                    "SHA-256 verification failed. "
                            + "Expected: " + expectedChecksum
                            + "; Actual: " + actualChecksum
            );

            throw new DocumentIntegrityException(
                    "Document integrity verification failed"
            );
        }

        // Record whether the user previewed or downloaded the file.
        auditService.record(
                username,
                auditAction,
                documentId,
                auditDetails
        );

        return new DocumentDownload(
                resource,
                document.getOriginalFilename(),
                document.getContentType()
        );
    }

    @Transactional
    public void restoreDocument(Long documentId, String username) {

        Document document = documentRepository.findById(documentId)
                .orElseThrow(() ->
                        new java.util.NoSuchElementException(
                                "Document not found"
                        )
                );

        if (document.getStatus() != DocumentStatus.DELETED) {
            throw new IllegalArgumentException(
                    "Only deleted documents can be restored"
            );
        }

        String markerVersionId =
                document.getDeleteMarkerVersionId();

        // Find the latest actual file version, excluding Delete Markers.
        StoredObjectVersion latestActualVersion =
                storageService.listVersions(document.getStorageKey())
                        .stream()
                        .filter(version -> !version.deleteMarker())
                        .findFirst()
                        .orElseThrow(() -> new IllegalStateException(
                                "No recoverable document version was found"
                        ));

        String sourceVersionId = latestActualVersion.versionId();

        // Read the metadata belonging to that exact S3 version.
        StoredObjectMetadata metadata =
                storageService.getVersionMetadata(
                        document.getStorageKey(),
                        sourceVersionId
                );

        // Load the selected historical version to calculate its checksum.
        Resource historicalResource;

        try {
            historicalResource = storageService.loadVersion(
                    document.getStorageKey(),
                    sourceVersionId
            );

        } catch (IOException exception) {

            throw new IllegalStateException(
                    "Unable to load the historical S3 document version",
                    exception
            );
        }
        String restoredChecksum;

        try (InputStream input =
                     historicalResource.getInputStream()) {

            MessageDigest digest =
                    MessageDigest.getInstance("SHA-256");

            byte[] buffer = new byte[8192];

            int bytesRead;

            while ((bytesRead = input.read(buffer)) != -1) {
                digest.update(buffer, 0, bytesRead);
            }

            restoredChecksum = HexFormat.of()
                    .formatHex(digest.digest());

        } catch (NoSuchAlgorithmException exception) {

            throw new IllegalStateException(
                    "SHA-256 algorithm is unavailable",
                    exception
            );

        } catch (IOException exception) {

            throw new IllegalStateException(
                    "Unable to read the recoverable document version",
                    exception
            );
        }

        // Restore the selected version and obtain the newly created S3 ID.
        String newVersionId =
                storageService.restoreVersionAndGetId(
                        document.getStorageKey(),
                        sourceVersionId
                );

        // Confirm the restored object is now accessible.
        if (!storageService.exists(document.getStorageKey())) {
            throw new IllegalStateException(
                    "Document recovery failed: object is not accessible"
            );
        }

        // Synchronize the current document metadata with the recovered file.
        document.setOriginalFilename(metadata.originalFilename());
        document.setContentType(metadata.contentType());
        document.setFileSizeBytes(metadata.sizeBytes());
        document.setSha256Checksum(restoredChecksum);
        document.setStatus(DocumentStatus.ACTIVE);
        document.setDeleteMarkerVersionId(null);

        Document savedDocument = documentRepository.save(document);

        // Record the newly created S3 version in MySQL.
        saveVersionMetadata(
                savedDocument,
                newVersionId,
                metadata.originalFilename(),
                metadata.contentType(),
                metadata.sizeBytes(),
                restoredChecksum
        );

        auditService.record(
                username,
                "DOCUMENT_RESTORE",
                documentId,
                "Document recovered from S3 version: "
                        + sourceVersionId
                        + "; new S3 version: "
                        + newVersionId
                        + "; previous Delete Marker: "
                        + markerVersionId
                        + "; SHA-256: "
                        + restoredChecksum
        );
    }

    // CHAPTER 6: S3-MYSQL VERSION RECONCILIATION

    @Transactional(readOnly = true)
    public VersionReconciliationResponse reconcileDocumentVersions(
            Long documentId) {

        // 1. Retrieve document from MySQL.

        Document document = documentRepository.findById(documentId)
                .orElseThrow(() ->
                        new java.util.NoSuchElementException(
                                "Document not found"
                        )
                );

        // 2. Retrieve actual S3 object versions.

        List<StoredObjectVersion> s3Versions =
                storageService.listVersions(
                        document.getStorageKey()
                );

        // 3. Retrieve version records from MySQL.

        List<DocumentVersion> databaseVersions =
                documentVersionRepository
                        .findByDocumentIdOrderByIdDesc(documentId);

        // 4. Separate actual S3 versions and Delete Markers.

        Map<String, StoredObjectVersion> s3VersionMap =
                new HashMap<>();

        List<String> deleteMarkerIds = new ArrayList<>();

        for (StoredObjectVersion version : s3Versions) {

            if (version.deleteMarker()) {

                deleteMarkerIds.add(version.versionId());

            } else {

                s3VersionMap.put(
                        version.versionId(),
                        version
                );
            }
        }

        // 5. Create a map of MySQL version records.

        Map<String, DocumentVersion> databaseVersionMap =
                new HashMap<>();

        for (DocumentVersion version : databaseVersions) {

            databaseVersionMap.put(
                    version.getS3VersionId(),
                    version
            );
        }

        // 6. Find database records missing in S3.

        List<String> missingInS3 = new ArrayList<>();

        for (String versionId : databaseVersionMap.keySet()) {

            if (!s3VersionMap.containsKey(versionId)) {

                missingInS3.add(versionId);
            }
        }

        // 7. Find S3 versions missing in MySQL.

        List<String> missingInDatabase = new ArrayList<>();

        for (String versionId : s3VersionMap.keySet()) {

            if (!databaseVersionMap.containsKey(versionId)) {

                missingInDatabase.add(versionId);
            }
        }

        // 8. Compare file sizes for matching version IDs.

        List<VersionReconciliationResponse.SizeMismatch>
                sizeMismatches = new ArrayList<>();

        int matchedVersionCount = 0;

        for (String versionId : s3VersionMap.keySet()) {

            DocumentVersion databaseVersion =
                    databaseVersionMap.get(versionId);

            if (databaseVersion != null) {

                matchedVersionCount++;

                long s3Size =
                        s3VersionMap.get(versionId).sizeBytes();

                long databaseSize =
                        databaseVersion.getFileSizeBytes();

                if (s3Size != databaseSize) {

                    sizeMismatches.add(
                            new VersionReconciliationResponse.SizeMismatch(
                                    versionId,
                                    s3Size,
                                    databaseSize
                            )
                    );
                }
            }
        }

        // 9. Determine reconciliation status.

        boolean consistent =
                missingInS3.isEmpty()
                        && missingInDatabase.isEmpty()
                        && sizeMismatches.isEmpty();

        String status = consistent
                ? "CONSISTENT"
                : "INCONSISTENT";

        String message = consistent
                ? "S3 and MySQL version histories are synchronized."
                : "Version inconsistencies detected between S3 and MySQL.";

        // 10. Return reconciliation report.

        return new VersionReconciliationResponse(

                document.getId(),

                document.getOriginalFilename(),

                status,

                s3VersionMap.size(),

                databaseVersionMap.size(),

                matchedVersionCount,

                missingInS3,

                missingInDatabase,

                sizeMismatches,

                deleteMarkerIds,

                message
        );
    }

// CHAPTER 7: HISTORICAL VERSION INTEGRITY VERIFICATION

    @Transactional(readOnly = true)
    public VersionIntegrityResponse verifyHistoricalVersionIntegrity(
            Long documentId) {

        // 1. Retrieve document from MySQL.

        Document document = documentRepository.findById(documentId)
                .orElseThrow(() ->
                        new java.util.NoSuchElementException(
                                "Document not found"
                        )
                );

        // 2. Retrieve all actual S3 versions.

        List<StoredObjectVersion> s3Versions =
                storageService.listVersions(
                        document.getStorageKey()
                );

        // 3. Retrieve all MySQL version records.

        List<DocumentVersion> databaseVersions =
                documentVersionRepository
                        .findByDocumentIdOrderByIdDesc(documentId);

        // 4. Separate S3 object versions and Delete Markers.

        Map<String, StoredObjectVersion> s3VersionMap =
                new HashMap<>();

        List<String> deleteMarkerIds = new ArrayList<>();

        for (StoredObjectVersion version : s3Versions) {

            if (version.deleteMarker()) {

                deleteMarkerIds.add(version.versionId());

            } else {

                s3VersionMap.put(
                        version.versionId(),
                        version
                );
            }
        }

        // 5. Build the MySQL version map.

        Map<String, DocumentVersion> databaseVersionMap =
                new HashMap<>();

        for (DocumentVersion version : databaseVersions) {

            databaseVersionMap.put(
                    version.getS3VersionId(),
                    version
            );
        }

        // 6. Identify versions missing from either system.

        List<String> missingInS3 = new ArrayList<>();

        for (String versionId : databaseVersionMap.keySet()) {

            if (!s3VersionMap.containsKey(versionId)) {
                missingInS3.add(versionId);
            }
        }

        List<String> missingInDatabase = new ArrayList<>();

        for (String versionId : s3VersionMap.keySet()) {

            if (!databaseVersionMap.containsKey(versionId)) {
                missingInDatabase.add(versionId);
            }
        }

        // 7. Verify the actual contents of every matching version.

        List<VersionIntegrityResponse.VersionCheck> checks =
                new ArrayList<>();

        int verifiedCount = 0;

        for (DocumentVersion databaseVersion : databaseVersions) {

            String versionId = databaseVersion.getS3VersionId();

            StoredObjectVersion s3Version =
                    s3VersionMap.get(versionId);

            // Version exists in MySQL but not in S3.

            if (s3Version == null) {

                checks.add(
                        new VersionIntegrityResponse.VersionCheck(
                                versionId,
                                "MISSING_IN_S3",
                                databaseVersion.getFileSizeBytes(),
                                null,
                                databaseVersion.getSha256Checksum(),
                                null,
                                "Database version has no corresponding S3 object version."
                        )
                );

                continue;
            }

            // 8. Retrieve the exact historical S3 version.

            try {

                org.springframework.core.io.Resource resource =
                        storageService.loadVersion(
                                document.getStorageKey(),
                                versionId
                        );

                // 9. Calculate actual SHA-256 and file size.

                MessageDigest digest =
                        MessageDigest.getInstance("SHA-256");

                long actualSize = 0;

                byte[] buffer = new byte[8192];

                try (InputStream input =
                             resource.getInputStream()) {

                    int bytesRead;

                    while ((bytesRead = input.read(buffer)) != -1) {

                        digest.update(buffer, 0, bytesRead);

                        actualSize += bytesRead;
                    }
                }

                String actualChecksum = HexFormat.of()
                        .formatHex(digest.digest());

                // 10. Compare actual values with MySQL.

                String expectedChecksum =
                        databaseVersion.getSha256Checksum();

                long expectedSize =
                        databaseVersion.getFileSizeBytes();

                boolean checksumMatches =
                        expectedChecksum.equalsIgnoreCase(
                                actualChecksum
                        );

                boolean sizeMatches =
                        expectedSize == actualSize;

                String verificationStatus;

                String message;

                if (checksumMatches && sizeMatches) {

                    verificationStatus = "VERIFIED";

                    message = "SHA-256 checksum and file size match.";

                    verifiedCount++;

                } else if (!checksumMatches && !sizeMatches) {

                    verificationStatus =
                            "CHECKSUM_AND_SIZE_MISMATCH";

                    message =
                            "Both SHA-256 checksum and file size differ.";

                } else if (!checksumMatches) {

                    verificationStatus = "CHECKSUM_MISMATCH";

                    message =
                            "Actual SHA-256 checksum differs from MySQL.";

                } else {

                    verificationStatus = "SIZE_MISMATCH";

                    message =
                            "Actual file size differs from MySQL.";
                }

                checks.add(
                        new VersionIntegrityResponse.VersionCheck(

                                versionId,

                                verificationStatus,

                                expectedSize,

                                actualSize,

                                expectedChecksum,

                                actualChecksum,

                                message
                        )
                );

            } catch (java.io.IOException exception) {

                checks.add(
                        new VersionIntegrityResponse.VersionCheck(

                                versionId,

                                "UNREADABLE",

                                databaseVersion.getFileSizeBytes(),

                                null,

                                databaseVersion.getSha256Checksum(),

                                null,

                                "Unable to read the historical S3 version."
                        )
                );

            } catch (NoSuchAlgorithmException exception) {

                throw new IllegalStateException(
                        "SHA-256 algorithm is unavailable",
                        exception
                );
            }
        }

        // 11. Determine overall integrity status.

        boolean allVerified =
                missingInS3.isEmpty()
                        && missingInDatabase.isEmpty()
                        && verifiedCount == databaseVersions.size();

        String integrityStatus = allVerified
                ? "VERIFIED"
                : "INTEGRITY_ISSUES_FOUND";

        String message = allVerified
                ? "All historical document versions passed integrity verification."
                : "One or more historical document versions failed integrity verification or could not be verified.";

        // 12. Return the final integrity report.

        return new VersionIntegrityResponse(

                document.getId(),

                document.getOriginalFilename(),

                integrityStatus,

                s3VersionMap.size(),

                databaseVersionMap.size(),

                verifiedCount,

                checks,

                missingInS3,

                missingInDatabase,

                deleteMarkerIds,

                message
        );
    }

                        new java.util.NoSuchElementException(

                sanitizeFilename(file.getOriginalFilename());

        String expectedContentType = ALLOWED_TYPES.get(extension);

        String actualContentType = file.getContentType();

        if (expectedContentType == null
                || !expectedContentType.equalsIgnoreCase(
                actualContentType == null ? "" : actualContentType
        )) {

            );

                file,
                extension
        );

            versionId = storageService.storeVersioned(
                    document.getStorageKey(),
                    inputStream,
                    expectedContentType,
                    originalFilename
            );
        }
        document.setOriginalFilename(originalFilename);
        document.setContentType(expectedContentType);
        document.setFileSizeBytes(file.getSize());
        document.setSha256Checksum(checksum);
        document.setStatus(DocumentStatus.ACTIVE);

        Document savedDocument = documentRepository.save(document);

        saveVersionMetadata(
                savedDocument,
                versionId,
                originalFilename,

                username,
                "DOCUMENT_NEW_VERSION_UPLOAD",

                        + "; SHA-256: " + checksum
        );

    }


    // Restore an existing historical S3 version.

    @Transactional
    public DocumentResponse restoreDocumentVersion(
            Long documentId,
            String versionId,
            String username) throws IOException {

        Document document = documentRepository.findById(documentId)
                .orElseThrow(() ->
                        new java.util.NoSuchElementException(
                                "Document not found"
                        )
                );

        boolean validVersion = storageService
                .listVersions(document.getStorageKey())
                .stream()
                .anyMatch(version ->
                        versionId.equals(version.versionId())
                                && !version.deleteMarker()
                );

        if (!validVersion) {
            throw new IllegalArgumentException(
                    "The requested historical version does not exist"
            );
        }

        // Retrieve the selected version's actual S3 metadata.
        StoredObjectMetadata metadata =
                storageService.getVersionMetadata(
                        document.getStorageKey(),
                        versionId
                );

        // Load the historical file and calculate its SHA-256 checksum.
        Resource historicalResource = storageService.loadVersion(
                document.getStorageKey(),
                versionId
        );

        String restoredChecksum;

        try (InputStream input =
                     historicalResource.getInputStream()) {

            MessageDigest digest =
                    MessageDigest.getInstance("SHA-256");

            byte[] buffer = new byte[8192];

            int bytesRead;

            while ((bytesRead = input.read(buffer)) != -1) {
                digest.update(buffer, 0, bytesRead);
            }

            restoredChecksum = HexFormat.of()
                    .formatHex(digest.digest());

        } catch (NoSuchAlgorithmException exception) {

            throw new IllegalStateException(
                    "SHA-256 algorithm is unavailable",
                    exception
            );
        }

        // Copy the selected historical version to create a new latest version.
        String newVersionId =
                storageService.restoreVersionAndGetId(
                        document.getStorageKey(),
                        versionId
                );

        // Update all current document metadata.
        document.setOriginalFilename(metadata.originalFilename());
        document.setContentType(metadata.contentType());
        document.setFileSizeBytes(metadata.sizeBytes());
        document.setSha256Checksum(restoredChecksum);
        document.setStatus(DocumentStatus.ACTIVE);

        Document savedDocument = documentRepository.save(document);

        // Record metadata for the newly created restored version.
        saveVersionMetadata(
                savedDocument,
                newVersionId,
                metadata.originalFilename(),
                metadata.contentType(),
                metadata.sizeBytes(),
                restoredChecksum
        );

        auditService.record(
                username,
                "DOCUMENT_VERSION_RESTORE",
                documentId,
                "Restored S3 version: " + versionId
                        + "; new S3 version: " + newVersionId
                        + "; SHA-256: " + restoredChecksum
        );

        return toResponse(savedDocument);
    }

    private void saveVersionMetadata(
            Document document,
            String versionId,
            String originalFilename,
            String contentType,
            long sizeBytes,
            String checksum) {

        // Local storage does not generate S3 version IDs.

        DocumentVersion version = new DocumentVersion();

        version.setDocument(document);
        version.setS3VersionId(versionId);
        version.setOriginalFilename(originalFilename);
        version.setContentType(contentType);
        version.setFileSizeBytes(sizeBytes);
        version.setSha256Checksum(checksum);

        documentVersionRepository.save(version);
    }

import org.springframework.security.oauth2.jwt.JwtClaimsSet;
import org.springframework.security.oauth2.jwt.JwtEncoder;
import org.springframework.security.oauth2.jwt.JwtEncoderParameters;
import org.springframework.security.oauth2.jwt.JwsHeader;
import org.springframework.stereotype.Service;

import java.time.Instant;
import java.util.List;

@Service
public class JwtService {

    private final JwtEncoder jwtEncoder;

    @Value("${app.jwt.issuer}")
    private String issuer;

    @Value("${app.jwt.expiration-minutes}")
    private long expirationMinutes;

    public JwtService(JwtEncoder jwtEncoder) {
        this.jwtEncoder = jwtEncoder;
    }

    public String generateToken(User user) {

        Instant now = Instant.now();

        Instant expiresAt = now.plusSeconds(
                expirationMinutes * 60
        );

        List<String> roles = user.getRoles()
                .stream()
                .map(role -> role.getName().name())
                .toList();

        JwtClaimsSet claims = JwtClaimsSet.builder()
                .issuer(issuer)
                .subject(user.getUsername())
                .issuedAt(now)
                .expiresAt(expiresAt)
                .claim("roles", roles)
                .build();

        JwsHeader header = JwsHeader
                .with(MacAlgorithm.HS256)
                .build();

        return jwtEncoder.encode(
                JwtEncoderParameters.from(header, claims)
        ).getTokenValue();
    }

    public long getExpirationSeconds() {
        return expirationMinutes * 60;
    }
}

    private JsonNode execute(Supplier<JsonNode> operation) {

        try {
            return operation.get();

        } catch (RestClientResponseException exception) {

            int code = exception.getStatusCode().value();

            HttpStatusCode mappedStatus;
            String message;

            if (code == 400 || code == 404 || code == 422) {
                mappedStatus = exception.getStatusCode();

                message = switch (code) {
                    case 400 -> "Python verifier rejected the request";
                    case 404 -> "Requested document or resource was not found";
                    case 422 -> "Invalid request parameters";
                    default -> "Python verifier request failed";
                };

            } else {
                // An upstream 401/403 indicates an internal
                // service-authentication problem, not a user login failure.
                mappedStatus = HttpStatus.BAD_GATEWAY;
                message = "Python verifier returned an unexpected response";
            }

            throw new PythonVerifierException(
                    mappedStatus,
                    message
            );

        } catch (ResourceAccessException exception) {

            throw new PythonVerifierException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Python verification service is unavailable or timed out"
            );
        }
    }

package com.evidencevault.backend.storage;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.core.io.FileSystemResource;
import org.springframework.core.io.Resource;
import org.springframework.stereotype.Service;

import java.io.FileNotFoundException;
import java.io.IOException;
import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardCopyOption;
import java.util.regex.Pattern;

@Service
@ConditionalOnProperty(
        name = "app.storage.provider",
        havingValue = "local",
        matchIfMissing = true
)
public class LocalFileStorageService implements StorageService {

    private static final Pattern STORAGE_KEY_PATTERN =
            Pattern.compile(
                    "^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$"
            );

    private final Path storageRoot;

    public LocalFileStorageService(
            @Value("${app.storage.local-root:./data/documents}")
            String storageLocation) {

        this.storageRoot = Path.of(storageLocation)
                .toAbsolutePath()
                .normalize();
    }

    @Override
    public void store(
            String storageKey,
            InputStream inputStream) throws IOException {

        Path destination = resolveSafePath(storageKey);

        Files.createDirectories(storageRoot);

        // Do not overwrite an existing stored object.
        try (InputStream input = inputStream) {
            Files.copy(input, destination);
        }
    }

    @Override
    public Resource load(String storageKey) throws IOException {

        Path filePath = resolveSafePath(storageKey);

        if (!Files.isRegularFile(filePath)) {
            throw new FileNotFoundException(
                    "Stored document was not found"
            );
        }

        return new FileSystemResource(filePath);
    }

    @Override
    public boolean exists(String storageKey) {

        try {
            Path filePath = resolveSafePath(storageKey);

            return Files.isRegularFile(filePath);

        } catch (IllegalArgumentException exception) {
            return false;
        }
    }

    private Path resolveSafePath(String storageKey) {

        if (storageKey == null
                || !STORAGE_KEY_PATTERN.matcher(storageKey).matches()) {

            throw new IllegalArgumentException(
                    "Invalid storage key"
            );
        }

        Path resolvedPath = storageRoot
                .resolve(storageKey)
                .normalize();

        if (!resolvedPath.startsWith(storageRoot)) {
            throw new IllegalArgumentException(
                    "Invalid storage path"
            );
        }

        return resolvedPath;
    }

    @Override
    public void delete(String storageKey) throws IOException {

        Path filePath = resolveSafePath(storageKey);

        Files.deleteIfExists(filePath);
    }
}

            String versionId) throws IOException {

        validateKey(storageKey);
        validateVersionId(versionId);

        GetObjectRequest request = GetObjectRequest.builder()
                .bucket(bucketName)
                .key(storageKey)
                .versionId(versionId)
                .build();

        ResponseBytes<GetObjectResponse> objectBytes =
                s3Client.getObjectAsBytes(request);

        return new ByteArrayResource(objectBytes.asByteArray());
    }

    @Override
    public boolean exists(String storageKey) {

        validateKey(storageKey);

        try {

            s3Client.headObject(
                    HeadObjectRequest.builder()
                            .bucket(bucketName)
                            .key(storageKey)
                            .build()
            );

            return true;

        } catch (NoSuchKeyException exception) {

            return false;

        } catch (S3Exception exception) {

            if (exception.statusCode() == 404) {
                return false;
            }

            throw exception;
        }
    }

    @Override
    public void delete(String storageKey) throws IOException {

        validateKey(storageKey);

        s3Client.deleteObject(
                DeleteObjectRequest.builder()
                        .bucket(bucketName)
                        .key(storageKey)
                        .build()
        );
    }

    @Override
    public String createDeleteMarker(String storageKey) {

        validateKey(storageKey);

        DeleteObjectResponse response = s3Client.deleteObject(
                DeleteObjectRequest.builder()
                        .bucket(bucketName)
                        .key(storageKey)
                        .build()
        );

        return response.versionId();
    }

    @Override
    public List<StoredObjectVersion> listVersions(
            String storageKey) {

        validateKey(storageKey);

        List<StoredObjectVersion> result = new ArrayList<>();

        String keyMarker = null;
        String versionIdMarker = null;

        boolean truncated;

        do {

            ListObjectVersionsRequest.Builder requestBuilder =
                    ListObjectVersionsRequest.builder()
                            .bucket(bucketName)
                            .prefix(storageKey)
                            .maxKeys(1000);

            if (keyMarker != null) {
                requestBuilder.keyMarker(keyMarker);
            }

            if (versionIdMarker != null) {
                requestBuilder.versionIdMarker(versionIdMarker);
            }

            ListObjectVersionsResponse response =
                    s3Client.listObjectVersions(
                            requestBuilder.build()
                    );

            for (ObjectVersion version : response.versions()) {

                if (storageKey.equals(version.key())) {

                    result.add(new StoredObjectVersion(
                            version.key(),
                            version.versionId(),
                            Boolean.TRUE.equals(version.isLatest()),
                            version.lastModified(),
                            version.size() == null ? 0L : version.size(),
                            version.eTag(),
                            false
                    ));
                }
            }

            for (DeleteMarkerEntry marker : response.deleteMarkers()) {

                if (storageKey.equals(marker.key())) {

                    result.add(new StoredObjectVersion(
                            marker.key(),
                            marker.versionId(),
                            Boolean.TRUE.equals(marker.isLatest()),
                            marker.lastModified(),
                            0L,
                            null,
                            true
                    ));
                }
            }

            truncated = Boolean.TRUE.equals(response.isTruncated());

            keyMarker = response.nextKeyMarker();
            versionIdMarker = response.nextVersionIdMarker();

        } while (truncated);

        result.sort(
                Comparator.comparing(
                        StoredObjectVersion::lastModified,
                        Comparator.nullsLast(Comparator.reverseOrder())
                )
        );

        return result;
    }

    @Override
    public void restoreVersion(
            String storageKey,
            String versionId
    ) {
        restoreVersionAndGetId(storageKey, versionId);
    }

    @Override
    public String restoreVersionAndGetId(
            String storageKey,
            String versionId
    ) {

        validateKey(storageKey);
        validateVersionId(versionId);

        StoredObjectVersion selectedVersion = listVersions(storageKey)
                .stream()
                .filter(version -> version.versionId().equals(versionId))
                .findFirst()
                .orElseThrow(() -> new IllegalArgumentException(
                        "Requested S3 version was not found"
                ));

        if (selectedVersion.deleteMarker()) {
            throw new IllegalArgumentException(
                    "Cannot restore a delete marker as a document"
            );
        }

        String encodedSource = URLEncoder.encode(
                storageKey,
                StandardCharsets.UTF_8
        ).replace("+", "%20");

        String encodedVersionId = URLEncoder.encode(
                versionId,
                StandardCharsets.UTF_8
        );

        String copySource = bucketName
                + "/"
                + encodedSource
                + "?versionId="
                + encodedVersionId;

        CopyObjectResponse response = s3Client.copyObject(
                CopyObjectRequest.builder()
                        .copySource(copySource)
                        .destinationBucket(bucketName)
                        .destinationKey(storageKey)
                        .metadataDirective(MetadataDirective.COPY)
                        .build()
        );

        return response.versionId();
    }

    @Override
    public void restoreLatestVersion(String storageKey) {
        restoreLatestVersionAndGetId(storageKey);
    }

    @Override
    public String restoreLatestVersionAndGetId(String storageKey) {

        validateKey(storageKey);

        StoredObjectVersion latestActualVersion = listVersions(storageKey)
                .stream()
                .filter(version -> !version.deleteMarker())
                .findFirst()
                .orElseThrow(() -> new IllegalStateException(
                        "No recoverable file version was found"
                ));

        return restoreVersionAndGetId(
                storageKey,
                latestActualVersion.versionId()
        );
    }

    private void validateKey(String storageKey) {

        try {

            UUID.fromString(storageKey);

        } catch (IllegalArgumentException exception) {

            throw new IllegalArgumentException(
                    "Invalid document storage key"
            );
        }
    }

    private void validateVersionId(String versionId) {

        if (versionId == null
                || versionId.isBlank()
                || "null".equalsIgnoreCase(versionId)) {

            throw new IllegalArgumentException(
                    "A valid S3 version ID is required"
            );
        }
    }

    @Override
    public StoredObjectMetadata getVersionMetadata(
            String storageKey,
            String versionId
    ) {

        validateKey(storageKey);
        validateVersionId(versionId);

        HeadObjectResponse response = s3Client.headObject(
                HeadObjectRequest.builder()
                        .bucket(bucketName)
                        .key(storageKey)
                        .versionId(versionId)
                        .build()
        );

        String originalFilename =
                response.metadata().get("original-filename");

        if (originalFilename == null || originalFilename.isBlank()) {
            originalFilename = storageKey;
        }

        String contentType = response.contentType();

        if (contentType == null || contentType.isBlank()) {
            contentType = "application/octet-stream";
        }

        return new StoredObjectMetadata(
                originalFilename,
                contentType,
                response.contentLength()
        );
    }
}

            String storageKey,

            String originalFilename) throws IOException {

        store(storageKey, inputStream);
    }

    boolean exists(String storageKey);

    void delete(String storageKey) throws IOException;

    // S3 version management operations

    default List<StoredObjectVersion> listVersions(
            String storageKey) {

        throw new UnsupportedOperationException(
                "Version listing is not supported by this storage provider"
        );
    }

    default Resource loadVersion(
            String storageKey,
            String versionId) throws IOException {

        throw new UnsupportedOperationException(
                "Version download is not supported by this storage provider"
        );
    }

    default void restoreVersion(
            String storageKey,
            String versionId) {

        throw new UnsupportedOperationException(
                "Version restoration is not supported by this storage provider"
        );
    }

    // Creates an S3 Delete Marker. Local storage keeps its existing
    // database-only soft-delete behaviour.

    default String createDeleteMarker(String storageKey)
            throws IOException {

        throw new UnsupportedOperationException(
                "Delete Markers are supported only by versioned storage"
        );
    }

    default void restoreLatestVersion(String storageKey) {

        if (!exists(storageKey)) {
            throw new IllegalStateException(
                    "No available document exists in storage"
            );
        }
    }

    default String storeVersioned(
            String storageKey,
            InputStream inputStream,
            String contentType,
            String originalFilename
    ) throws IOException {
        store(storageKey, inputStream, contentType, originalFilename);
        return null;
    }

            String storageKey,

        throw new UnsupportedOperationException(
                "Version metadata is not supported by this storage provider"
        );

            String storageKey,
            String versionId
    ) {
        restoreVersion(storageKey, versionId);
        return null;
    }

    default String restoreLatestVersionAndGetId(String storageKey) {
        restoreLatestVersion(storageKey);
        return null;
    }
}
