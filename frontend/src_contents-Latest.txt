import { BrowserRouter, Routes, Route } from "react-router-dom";

import { ThemeProvider } from "./context/ThemeContext";
import { AuthProvider } from "./context/AuthContext";

import { USER_ROLES } from "./config/constants";

import Layout from "./components/Layout";
import ProtectedRoute from "./components/ProtectedRoute";

import Home from "./pages/Home";
import Login from "./pages/Login";
import Dashboard from "./pages/Dashboard";
import Documents from "./pages/Documents";
import Recovery from "./pages/Recovery";
import Audit from "./pages/Audit";
import Unauthorized from "./pages/Unauthorized";

function App() {
  return (
    <ThemeProvider>
      <AuthProvider>
        <BrowserRouter>
          <Routes>
            <Route element={<Layout />}>
              {/* Public pages */}
              <Route path="/" element={<Home />} />
              <Route path="/login" element={<Login />} />
              <Route path="/unauthorized" element={<Unauthorized />} />

              {/* All authenticated users */}
              <Route element={<ProtectedRoute />}>
                <Route path="/dashboard" element={<Dashboard />} />

                <Route path="/documents" element={<Documents />} />
              </Route>

              {/* Admin and Operator */}
              <Route
                element={
                  <ProtectedRoute
                    allowedRoles={[USER_ROLES.ADMIN, USER_ROLES.OPERATOR]}
                  />
                }
              >
                <Route path="/recovery" element={<Recovery />} />
              </Route>

              {/* Admin and Auditor */}
              <Route
                element={
                  <ProtectedRoute
                    allowedRoles={[USER_ROLES.ADMIN, USER_ROLES.AUDITOR]}
                  />
                }
              >
                <Route path="/audit" element={<Audit />} />
              </Route>
            </Route>
          </Routes>
        </BrowserRouter>
      </AuthProvider>
    </ThemeProvider>
  );
}

export default App;

function ProtectedRoute({ allowedRoles = [] }) {
  const { isAuthenticated, hasRole } = useAuth();

  const location = useLocation();

  if (!isAuthenticated) {
    return <Navigate to="/login" replace state={{ from: location }} />;
  }

import { createContext, useContext, useEffect, useState } from "react";

import {
  loginUser,
  saveAuthSession,
  getStoredUser,
  clearAuthSession,
} from "../services/authService";

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(() => getStoredUser());

  const [isAuthenticated, setIsAuthenticated] = useState(() =>
    Boolean(localStorage.getItem("evidence-vault-token") && getStoredUser()),
  );

  const [loading, setLoading] = useState(false);

  // Restore the saved session on application startup.
  useEffect(() => {
    const storedUser = getStoredUser();
    const storedToken = localStorage.getItem("evidence-vault-token");

    if (storedUser && storedToken) {
      setUser(storedUser);
      setIsAuthenticated(true);
    } else {
      clearAuthSession();
      setUser(null);
      setIsAuthenticated(false);
    }
  }, []);

  const login = async (credentials) => {
    setLoading(true);

    try {
      const authData = await loginUser(credentials);

      if (!authData?.accessToken) {
        throw new Error("Authentication token was not received.");
      }

      saveAuthSession(authData);

      setUser({
        username: authData.username,
        roles: authData.roles || [],
        tokenType: authData.tokenType,
        expiresIn: authData.expiresIn,
      });

      setIsAuthenticated(true);

      return authData;
    } finally {
      setLoading(false);
    }
  };

  const logout = () => {
    clearAuthSession();

    setUser(null);
    setIsAuthenticated(false);
  };

  const hasRole = (allowedRoles = []) => {
    if (!user || !Array.isArray(user.roles)) {
      return false;
    }

    return allowedRoles.some((role) => user.roles.includes(role));
  };

  return (
    <AuthContext.Provider
      value={{
        user,
        isAuthenticated,
        loading,
        login,
        logout,
        hasRole,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);

  if (!context) {
    throw new Error("useAuth must be used inside AuthProvider");
  }

  return context;
}

  name,
  verification,
  verifiedAt,
  secondary = false,
}) {
  const hasResult = verification === true || verification === false;

  return (
    <article className="dash-region-card">
      <div className="dash-region-top">
        <div className={`dash-region-flag ${secondary ? "singapore" : ""}`}>
          {region}
        </div>

        <div className="dash-region-info">
          <strong>{name}</strong>
          <span>{code}</span>
        </div>

        <span className="dash-region-type">
          {secondary ? "REPLICA" : "PRIMARY"}
        </span>
      </div>

      <div className="dash-region-bottom">
        <div>
          <span>Last recorded verification</span>
          <strong>
            {hasResult
              ? verification
                ? "Checksum verified"
                : "Verification failed"
              : "No verification recorded"}
          </strong>
        </div>

        <StatusPill
          status={
            hasResult ? (verification ? "VERIFIED" : "FAILED") : "UNKNOWN"
          }
          label={
            hasResult ? (verification ? "Verified" : "Failed") : "Not checked"
          }
        />
      </div>

      <small className="dash-region-date">
        {verifiedAt ? formatDate(verifiedAt) : "Awaiting first recorded drill"}
      </small>
    </article>
  );
}

function Dashboard() {
  const { hasRole, user } = useAuth();

  const canViewRecovery = hasRole([USER_ROLES.ADMIN, USER_ROLES.AUDITOR]);

  const [dashboard, setDashboard] = useState(null);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [pageError, setPageError] = useState("");
  const [lastUpdated, setLastUpdated] = useState(null);

  const loadDashboard = useCallback(
    async (isRefresh = false) => {
      if (isRefresh) {
        setRefreshing(true);
      } else {
        setLoading(true);
      }

      setPageError("");

      try {
        const result = await getDashboardData({
          includeRecoveryData: canViewRecovery,
        });

        setDashboard(result);
        setLastUpdated(new Date());
      } catch (error) {
        setPageError(error?.message || "Unable to load dashboard information.");
      } finally {
        setLoading(false);
        setRefreshing(false);
      }
    },
    [canViewRecovery],
  );

  useEffect(() => {
    loadDashboard();
  }, [loadDashboard]);

  const documents = dashboard?.documents || {
    total: 0,
    active: 0,
    deleted: 0,
    corrupted: 0,
    recent: [],
  };

  const metrics = useMemo(() => toArray(dashboard?.metrics), [dashboard]);

  const drills = useMemo(() => toArray(dashboard?.drills), [dashboard]);

  const manifests = useMemo(() => toArray(dashboard?.manifests), [dashboard]);

  const recentDocuments = useMemo(() => {
    return [...toArray(documents.recent)].sort(
      (a, b) => new Date(b.createdAt) - new Date(a.createdAt),
    );
  }, [documents.recent]);

  const latestDrill = useMemo(() => getLatestRecord(drills), [drills]);

  const latestManifest = useMemo(() => getLatestRecord(manifests), [manifests]);

  const drillSummary = useMemo(() => {
    const successful = drills.filter((item) =>
      ["SUCCESS", "COMPLETED"].includes(normalizeStatus(item.drill_status)),
    ).length;

    const failed = drills.filter(
      (item) => normalizeStatus(item.drill_status) === "FAILED",
    ).length;

    const attention = drills.filter((item) =>
      ["RECOVERY_REQUIRED", "WARNING"].includes(
        normalizeStatus(item.drill_status),
      ),
    ).length;

    return {
      total: drills.length,
      successful,
      failed,
      attention,
      successRate: drills.length
        ? Number(((successful / drills.length) * 100).toFixed(1))
        : null,
      recoveryOperations: drills.filter(
        (item) => item.recovery_performed === true,
      ).length,
    };
  }, [drills]);

  const metricSummary = useMemo(() => {
    const rtoValues = metrics
      .map((item) => Number(item.recovery_workflow_seconds))
      .filter(Number.isFinite);

    const rpoValues = metrics
      .map((item) => Number(item.estimated_rpo_seconds))
      .filter(Number.isFinite);

    const average = (values) =>
      values.length
        ? values.reduce((sum, value) => sum + value, 0) / values.length
        : null;

    return {
      averageRto: average(rtoValues),
      averageRpo: average(rpoValues),
      replicationCompleted: metrics.filter(
        (item) => item.replication_status === "COMPLETED",
      ).length,
      replicationFailed: metrics.filter(
        (item) => item.replication_status === "FAILED",
      ).length,
      replicationObserved: metrics.filter(
        (item) => item.replication_status !== "NOT_AVAILABLE",
      ).length,
    };
  }, [metrics]);

  const rtoChartData = useMemo(() => {
    return [...metrics]
      .filter((item) => Number.isFinite(Number(item.recovery_workflow_seconds)))
      .sort((a, b) => new Date(a.created_at) - new Date(b.created_at))
      .slice(-10)
      .map((item, index) => ({
        name: `Drill ${index + 1}`,
        rto: Number(item.recovery_workflow_seconds),
        date: formatDate(item.created_at),
      }));
  }, [metrics]);

  const drillChartData = useMemo(() => {
    return [
      { name: "Successful", value: drillSummary.successful },
      { name: "Failed", value: drillSummary.failed },
      { name: "Attention", value: drillSummary.attention },
    ].filter((item) => item.value > 0);
  }, [drillSummary]);

  const system = dashboard?.system;

  const systemHealthy =
    system?.application === "UP" && system?.database === "UP";

  const latestDrillDate = latestDrill?.created_at;

  if (loading) {
    return (
      <main className="dashboard-page">
        <div className="dash-loading">
          <RefreshCw className="spin" size={25} />
          <strong>Preparing your EvidenceVault overview</strong>
          <span>
            Collecting document, infrastructure and recovery information.
          </span>
        </div>
      </main>
    );
  }

  if (pageError && !dashboard) {
    return (
      <main className="dashboard-page">
        <div className="dash-error">
          <AlertTriangle size={30} />
          <h2>Dashboard unavailable</h2>
          <p>{pageError}</p>
          <button onClick={() => loadDashboard()}>
            <RefreshCw size={16} />
            Try again
          </button>
        </div>
      </main>
    );
  }

  return (
    <main className="dashboard-page">
      {/* PAGE HEADER */}
      <header className="dash-page-header">
        <div>
          <span className="dash-eyebrow">EVIDENCE VAULT / OVERVIEW</span>

            Your document protection, infrastructure and recovery operations at
            a glance.
          </p>
        </div>

          <div className="dash-updated">

                    minute: "2-digit",

            </span>
          </div>
        </div>

        <div className="dash-system-banner-status">
          <StatusPill

            label={`Application ${system?.application || "Unknown"}`}
          />
          <StatusPill
            status={system?.database || "UNKNOWN"}
            label={`Database ${system?.database || "Unknown"}`}
          />
        </div>
      </section>

      {/* DOCUMENT KPI CARDS */}
      <section className="dash-section">
        <SectionHeading
          eyebrow="DOCUMENT LIFECYCLE"
          title="Document overview"
          description="Understand how many documents are stored and their current lifecycle state."

            <Link to="/documents" className="dash-text-link">

            </Link>
          }
        />

        <div className="dash-metric-grid">
          <MetricCard

            tone="blue"
            to="/documents"
          />

          <MetricCard
            title="Active documents"
            value={formatNumber(documents.active)}

            tone="green"
            to="/documents"
          />

          <MetricCard

            icon={Trash2}
            tone="orange"

          />

          <MetricCard

      </section>

      {/* RECOVERY KPI CARDS */}
      <section className="dash-section">
        <SectionHeading
          eyebrow="BUSINESS CONTINUITY"
          title="Disaster recovery performance"
          description="Measured results from previously recorded recovery operations."

            canViewRecovery ? (

          }
        />

        {canViewRecovery ? (
          <>

              <MetricCard

                description="Recorded operations"
                icon={Activity}

              />

                title="Drill success rate"

              />

                title="Average recovery time"

                tone="purple"
              />

              <MetricCard
                title="Estimated RPO"

                tone="orange"
              />

              <MetricCard

                tone="green"
              />
            </div>

            {dashboard?.errors?.metrics && (

              <p className="dash-section-error">

          </>
        ) : (
          <div className="dash-permission-note">
            <ShieldCheck size={21} />
            <div>

              <span>
                Detailed drill history, RTO and recovery evidence are available
                to Admin and Auditor accounts.
              </span>
            </div>
          </div>
        )}
      </section>

      {/* ANALYTICS AND SYSTEM HEALTH */}
      <section className="dash-content-grid">
        <div className="dash-panel">
          <SectionHeading
            eyebrow="PERFORMANCE"
            title="Recovery time trend"
            description="Recovery workflow duration from recorded metrics."
          />

          {canViewRecovery && rtoChartData.length > 0 ? (
            <div className="dash-chart">
              <ResponsiveContainer width="100%" height={260}>
                <AreaChart data={rtoChartData}>
                  <defs>
                    <linearGradient id="rtoFill" x1="0" y1="0" x2="0" y2="1">
                      <stop
                        offset="0%"
                        stopColor="#6385f5"
                        stopOpacity={0.35}

                      <stop

                        stopColor="#6385f5"
                        stopOpacity={0.02}

                    </linearGradient>
                  </defs>

                  <CartesianGrid
                    stroke="var(--border-color)"
                    strokeDasharray="4 4"
                    vertical={false}
                  />

                  <XAxis
                    dataKey="name"
                    tick={{ fill: "var(--text-muted)", fontSize: 11 }}
                    axisLine={false}
                    tickLine={false}
                  />

                  <YAxis
                    tick={{ fill: "var(--text-muted)", fontSize: 11 }}
                    axisLine={false}
                    tickLine={false}

                  />

                      "Recovery time",
                    ]}
                    labelFormatter={(label, payload) =>
                      payload?.[0]?.payload?.date || label
                    }
                  />

                  <Area
                    type="monotone"
                    dataKey="rto"
                    stroke="#6385f5"
                    strokeWidth={2.5}
                    fill="url(#rtoFill)"
                    activeDot={{ r: 5 }}
                  />

            </div>

              title="No recovery measurements yet"
              description={
                canViewRecovery
                  ? "Run a recovery drill to start building your RTO history."
                  : "Recovery measurements are restricted to Admin and Auditor roles."
              }
            />

            <>

                <ResponsiveContainer width="100%" height={220}>
                  <PieChart>
                    <Pie
                      data={drillChartData}
                      dataKey="value"
                      nameKey="name"
                      innerRadius={57}
                      outerRadius={83}

                        />

                        border: "1px solid var(--border-color)",
                        borderRadius: 10,
                        color: "var(--text-primary)",
                      }}
                    />

                    <Legend
                      verticalAlign="bottom"
                      iconType="circle"
                      formatter={(value) => (
                        <span className="dash-chart-legend">{value}</span>
                      )}
                    />
                  </PieChart>
                </ResponsiveContainer>
              </div>

              <div className="dash-outcome-summary">
                <div>
                  <span>Successful</span>
                  <strong className="text-success">
                    {drillSummary.successful}
                  </strong>
                </div>
                <div>
                  <span>Failed</span>
                  <strong className="text-danger">{drillSummary.failed}</strong>
                </div>
                <div>
                  <span>Attention</span>
                  <strong className="text-warning">
                    {drillSummary.attention}
                  </strong>
                </div>
              </div>
            </>

            <EmptyState
              icon={Activity}
              title="No drill outcomes recorded"

            <div className="dash-connector-line" />
            <div className="dash-connector-icon">
              <ArrowRight size={18} />

          />

            <div className="dash-replication-card">

                <span>Completed observations</span>
                <strong>{metricSummary.replicationCompleted}</strong>

                <span>Recorded observations</span>

                          <StatusPill

                                : drill.secondary_verified === false
                                  ? "FAILED"
                                  : "UNKNOWN"
                            }
                          />
                        </td>

                        <td>
                          <StatusPill status={drill.drill_status} />
                        </td>

                        <td>
                          {formatSeconds(

function DocumentIcon({ contentType }) {
  if (contentType?.startsWith("image/")) {
    return <FileImage size={20} />;
  }

  if (contentType === "application/pdf") {
    return <FileText size={20} />;
  }

  return <File size={20} />;
}

// --------------------------------------------------
// Document status
// --------------------------------------------------

function DocumentStatus({ status }) {
  const value = getDocumentStatus(status);

  const className =
    value === "ACTIVE" || value === "AVAILABLE"
      ? "document-status-active"
      : value === "DELETED"
        ? "document-status-deleted"
        : "document-status-neutral";

  return (
    <span className={`document-status ${className}`}>
      {value.replaceAll("_", " ")}
    </span>
  );
}

export default function Documents() {
  // Document data
  const [documents, setDocuments] = useState([]);
  const [page, setPage] = useState(0);
  const [totalPages, setTotalPages] = useState(0);
  const [totalElements, setTotalElements] = useState(0);

  // Draft filters: values currently entered by the user
  const [keyword, setKeyword] = useState("");
  const [category, setCategory] = useState("");
  const [status, setStatus] = useState("");

  // Applied filters: values actually sent to the backend
  const [appliedFilters, setAppliedFilters] = useState({
    ...EMPTY_FILTERS,
  });

  // Loading and action states
  const [loading, setLoading] = useState(true);
  const [uploading, setUploading] = useState(false);
  const [actionId, setActionId] = useState(null);

  // Modal states
  const [showUpload, setShowUpload] = useState(false);

  const [selectedPreviewDocument, setSelectedPreviewDocument] = useState(null);

  const [previewUrl, setPreviewUrl] = useState("");
  const [previewText, setPreviewText] = useState("");

  // Alert states
  const [message, setMessage] = useState("");
  const [error, setError] = useState("");

  // Tracks the latest list request.
  // This prevents an older response from replacing newer data.
  const requestSequence = useRef(0);

  // --------------------------------------------------
  // Load documents
  // --------------------------------------------------

  const loadDocuments = useCallback(
    async (overrides = {}) => {
      const requestId = ++requestSequence.current;

      const requestPage = overrides.page ?? page;

      const requestFilters = overrides.filters ?? appliedFilters;

      setLoading(true);
      setError("");

      try {
        const result = await getDocuments({
          keyword: requestFilters.keyword,
          category: requestFilters.category,
          status: requestFilters.status,
          page: requestPage,
          size: PAGE_SIZE,
        });

        // Ignore an outdated response.
        if (requestId !== requestSequence.current) {
          return;
        }

        setDocuments(Array.isArray(result?.content) ? result.content : []);

        setTotalPages(result?.totalPages ?? 0);
        setTotalElements(result?.totalElements ?? 0);
      } catch (err) {
        if (requestId !== requestSequence.current) {
          return;
        }

        setError(err?.message || "Unable to load documents.");

        setDocuments([]);
        setTotalPages(0);
        setTotalElements(0);
      } finally {
        if (requestId === requestSequence.current) {
          setLoading(false);
        }
      }
    },
    [page, appliedFilters],
  );

  // Automatically load when the page or applied filters change.
  useEffect(() => {
    loadDocuments();
  }, [loadDocuments]);

  // --------------------------------------------------
  // Preview URL cleanup
  // --------------------------------------------------

  useEffect(() => {
    return () => {
      if (previewUrl) {
        URL.revokeObjectURL(previewUrl);
      }
    };
  }, [previewUrl]);

  // --------------------------------------------------
  // Active documents summary
  // --------------------------------------------------

  const activeDocuments = useMemo(() => {
    return documents.filter(
      (item) => getDocumentStatus(item.status) !== "DELETED",
    );
  }, [documents]);

  // --------------------------------------------------
  // Upload document
  // --------------------------------------------------

  async function handleUpload(file, selectedCategory, description) {
    setUploading(true);
    setError("");
    setMessage("");

    try {
      await uploadDocument(file, selectedCategory, description);

      setShowUpload(false);

      // Reset the filters after a successful upload.
      setKeyword("");
      setCategory("");
      setStatus("");

      setAppliedFilters({
        ...EMPTY_FILTERS,
      });

      setPage(0);

      setMessage("Document uploaded successfully.");

      // Explicitly refresh page 0 with empty filters.
      await loadDocuments({
        page: 0,
        filters: EMPTY_FILTERS,
      });
    } catch (err) {
      setError(err?.message || "Document upload failed.");
    } finally {
      setUploading(false);
    }
  }

  // --------------------------------------------------
  // Download document
  // --------------------------------------------------

  async function handleDownload(document) {
    setActionId(document.id);
    setError("");
    setMessage("");

    let downloadUrl = "";

    try {
      const blob = await downloadDocument(document.id);

      if (!(blob instanceof Blob) || blob.size === 0) {
        throw new Error("The downloaded document is empty.");
      }

      downloadUrl = URL.createObjectURL(blob);

      const link = window.document.createElement("a");

      link.href = downloadUrl;
      link.download = document.originalFilename || "document";

      window.document.body.appendChild(link);
      link.click();
      link.remove();

      setMessage("Document download started.");
    } catch (err) {
      setError(err?.message || "Download failed.");
    } finally {
      if (downloadUrl) {
        window.setTimeout(() => {
          URL.revokeObjectURL(downloadUrl);
        }, 1000);
      }

      setActionId(null);
    }
  }

  // --------------------------------------------------
  // Preview document
  // --------------------------------------------------

  async function handlePreview(document) {
    if (!isPreviewable(document.contentType, document.originalFilename)) {
      setError(
        "Preview is not supported for this file type. You can download the document instead.",
      );

      return;
    }

    setActionId(document.id);
    setError("");
    setMessage("");

    try {
      const blob = await downloadDocument(document.id);

      if (!(blob instanceof Blob) || blob.size === 0) {
        throw new Error("The preview response is empty.");
      }

      const contentType = document.contentType?.toLowerCase() || "";

      const isText =
        contentType.startsWith("text/") ||
        contentType === "application/json" ||
        contentType === "application/xml";

      // Text and CSV files are displayed as text.
      if (isText) {
        const text = await blob.text();

        setPreviewUrl("");
        setPreviewText(text);
        setSelectedPreviewDocument(document);
      } else {
        // PDF and image files use a temporary object URL.
        const url = URL.createObjectURL(blob);

        setPreviewText("");
        setPreviewUrl(url);
        setSelectedPreviewDocument(document);
      }
    } catch (err) {
      setError(err?.message || "Unable to preview document.");
    } finally {
      setActionId(null);
    }
  }

  // --------------------------------------------------
  // Close preview
  // --------------------------------------------------

  function closePreview() {
    setSelectedPreviewDocument(null);
    setPreviewUrl("");
    setPreviewText("");
  }

  // --------------------------------------------------
  // Delete document
  // --------------------------------------------------

  async function handleDelete(document) {
    const confirmed = window.confirm(
      `Are you sure you want to delete "${document.originalFilename}"?`,
    );

    if (!confirmed) {
      return;
    }

    setActionId(document.id);
    setError("");
    setMessage("");

    try {
      await deleteDocument(document.id);

      setMessage("Document deleted successfully.");

      // If the deleted document was the only record
      // on the current page, move to the previous page.
      if (documents.length === 1 && page > 0) {
        setPage((current) => current - 1);
      } else {
        await loadDocuments();
      }
    } catch (err) {
      setError(err?.message || "Unable to delete document.");
    } finally {
      setActionId(null);
    }
  }

  // --------------------------------------------------
  // Restore document
  // --------------------------------------------------

  async function handleRestore(document) {
    setActionId(document.id);
    setError("");
    setMessage("");

    try {
      await restoreDocument(document.id);

      setMessage("Document restored successfully.");

      await loadDocuments();
    } catch (err) {
      setError(err?.message || "Unable to restore document.");
    } finally {
      setActionId(null);
    }
  }

  // --------------------------------------------------
  // Search and filters
  // --------------------------------------------------

  function applySearch(event) {
    event.preventDefault();

    setAppliedFilters({
      keyword: keyword.trim(),
      category: category.trim(),
      status,
    });

    setPage(0);
  }

  function clearFilters() {
    setKeyword("");
    setCategory("");
    setStatus("");

    setAppliedFilters({
      ...EMPTY_FILTERS,
    });

    setPage(0);
  }

  // --------------------------------------------------
  // Refresh
  // --------------------------------------------------

  function handleRefresh() {
    loadDocuments();
  }

  // --------------------------------------------------
  // Render
  // --------------------------------------------------

  return (
    <main className="documents-page">
      {/* Page heading */}

      <section className="documents-heading">
        <div>
          <span className="documents-eyebrow">EVIDENCE VAULT</span>

          <h1>Document Management</h1>

          <p>Upload, organize, preview and manage your protected documents.</p>
        </div>

        <button
          className="document-primary-button"
          onClick={() => setShowUpload(true)}
          type="button"
        >
          <Upload size={17} />
          Upload document
        </button>
      </section>

      {/* Summary cards */}

      <section className="document-summary-grid">
        <div className="document-summary-card">
          <div className="document-summary-icon">
            <FolderOpen size={21} />
          </div>

          <div>
            <span>Total records</span>
            <strong>{totalElements}</strong>
          </div>
        </div>

        <div className="document-summary-card">
          <div className="document-summary-icon summary-green">
            <CheckCircle2 size={21} />
          </div>

          <div>
            <span>Active on this page</span>
            <strong>{activeDocuments.length}</strong>
          </div>

        <div className="document-summary-card">
          <div className="document-summary-icon summary-purple">

async function readResponseBody(response) {
  const contentType = response.headers.get("content-type") || "";

  try {
    // Handles application/json and application/problem+json.
    if (contentType.toLowerCase().includes("json")) {
      return await response.json();
    }

    return await response.text();
  } catch {
    return null;
  }
}

function extractMessage(data) {
  if (!data) return "";

  if (typeof data === "string") {
    const trimmed = data.trim();

    // Avoid showing a JSON response as raw text.
    if (trimmed.startsWith("{") || trimmed.startsWith("[")) {
      try {
        return extractMessage(JSON.parse(trimmed));
      } catch {
        return trimmed;
      }
    }

    return trimmed;
  }

  if (Array.isArray(data)) {
    return data
      .map((item) => {
        if (typeof item === "string") return item;
        return item?.defaultMessage || item?.message || "";
      })
      .filter(Boolean)
      .join(", ");
  }

  if (typeof data === "object") {
    const detail = data.detail;

    if (typeof detail === "string") return detail;

    if (Array.isArray(detail)) {
      return extractMessage(detail);
    }

    return (
      data.message ||
      data.error ||
      data.title ||
      ""
    );
  }

  return "";
}

export function createApiErrorMessage(
  response,
  responseData,
  endpoint = "",
  customFallback = ""
) {
  const status = response.status;

  if (status === 401) {
    if (endpoint.includes("/auth/login")) {
      return "Invalid username or password. Please check your credentials and try again.";
    }

    return "Your session has expired. Please sign in again.";
  }

  if (status === 403) {
    return "You do not have permission to perform this action.";
  }

  if (status === 404) {
    return "The requested resource could not be found.";
  }

  if (status === 409) {
    return "This operation cannot be completed because the resource is in its current state.";
  }

  if (status === 413) {
    return "The selected file is too large to upload.";
  }

  if (status === 415) {
    return "This file format is not supported by the server.";
  }

  if (status === 422) {
    return (
      extractMessage(responseData) ||
      "Some of the submitted information is invalid. Please check your inputs."
    );
  }

  if (status >= 500) {
    return (
      customFallback ||
      "The server encountered an error. Please try again later."
    );
  }

  return (
    extractMessage(responseData) ||
    customFallback ||
    "The request could not be completed. Please try again."
  );
}

export async function getApiErrorMessage(
  response,
  fallback = "",
  endpoint = ""
) {
  const responseData = await readResponseBody(response);

  return createApiErrorMessage(
    response,
    responseData,
    endpoint,
    fallback
  );
}

export async function apiRequest(endpoint, options = {}) {
  const token = localStorage.getItem("evidence-vault-token");

  const headers = {
    ...options.headers,
  };

  const isFormData = options.body instanceof FormData;

  // JSON requests need Content-Type.
  // FormData must NOT have a manually assigned Content-Type:
  // the browser adds the multipart boundary automatically.
  if (
    options.body !== undefined &&
    !isFormData &&
    !headers["Content-Type"]
  ) {
    headers["Content-Type"] = "application/json";
  }

  // Do not attach an old JWT to the login request.
  if (token && !endpoint.includes("/auth/login")) {
    headers.Authorization = `Bearer ${token}`;
  }

  let response;

  try {
    response = await fetch(`${API_BASE_URL}${endpoint}`, {
      ...options,
      headers,
    });
  } catch {
    throw new Error(
      "Unable to connect to the server. Check that the backend is running and try again."
    );
  }

  if (response.status === 204) {
    return null;
  }

  const responseData = await readResponseBody(response);

  if (!response.ok) {
    const message = createApiErrorMessage(
      response,
      responseData,
      endpoint
    );

    throw new Error(message);
  }

  return responseData;
}

export function average(values) {
  const valid = values
    .filter((value) => value !== null && value !== undefined && value !== "")
    .map(Number)
    .filter(Number.isFinite);

  if (!valid.length) return null;

  return Number(
    (valid.reduce((sum, value) => sum + value, 0) / valid.length).toFixed(3),
  );
}

export function calculateDrillSummary(drills) {
  const total = drills.length;

  const successful = drills.filter(
    (item) => item.drill_status === "SUCCESS",
  ).length;

  const failed = drills.filter((item) => item.drill_status === "FAILED").length;

  const attentionRequired = drills.filter(
    (item) => item.drill_status === "RECOVERY_REQUIRED",
  ).length;

  const recoveryOperations = drills.filter(
    (item) => item.recovery_performed === true,
  ).length;

  const corruptionDrills = drills.filter(
    (item) =>
      item.corruption_detected === true ||
      item.failure_type === "SIMULATED_CORRUPTION",
  ).length;

  const primaryFailures = drills.filter(
    (item) => item.failure_type === "SIMULATED_PRIMARY_FAILURE",
  ).length;

  const checksumMismatches = drills.filter((item) => {
    const primaryMismatch =
      item.primary_checksum &&
      item.expected_checksum &&
      item.primary_checksum !== item.expected_checksum;

    const secondaryMismatch =
      item.secondary_checksum &&
      item.expected_checksum &&
      item.secondary_checksum !== item.expected_checksum;

    return primaryMismatch || secondaryMismatch;
  }).length;

  return {
    total,
    successful,
    failed,
    attentionRequired,
    recoveryOperations,
    corruptionDrills,
    primaryFailures,
    checksumMismatches,

    successRate: total ? Number(((successful / total) * 100).toFixed(1)) : null,
  };
}

export function calculateMetricSummary(metrics) {
  const rtoValues = metrics.map(
    (item) => item.recovery_rto_seconds ?? item.recovery_workflow_seconds,
  );

  const rpoValues = metrics
    .map((item) => item.estimated_rpo_seconds)
    .filter((value) => value !== null && value !== undefined);

  const replicationCompleted = metrics.filter(
    (item) => item.replication_status === "COMPLETED",
  ).length;

  const replicationFailed = metrics.filter(
    (item) => item.replication_status === "FAILED",
  ).length;

  const replicationTimedOut = metrics.filter(
    (item) => item.replication_timed_out === true,
  ).length;

  return {
    averageRto: average(rtoValues),
    averageRpo: average(rpoValues),
    replicationCompleted,
    replicationFailed,
    replicationTimedOut,
  };
}

function getFriendlyError(status) {
  switch (status) {
    case 401:
      return "Your session has expired. Please sign in again.";
    case 403:
      return "You don't have permission to access this document.";
    case 404:
      return "The document could not be found.";
    case 500:
      return "The server couldn't retrieve this document. Please try again.";
    default:
      return "Unable to retrieve the document. Please try again.";
  }
}

export async function fetchDocumentBlob(documentId) {
  const token = localStorage.getItem("evidence-vault-token");

  if (!token) {
    throw new Error("Your session has expired. Please sign in again.");
  }

  const response = await fetch(
    `${API_BASE_URL}/v1/documents/${documentId}/download`,
    {
      method: "GET",
      headers: {
        Authorization: `Bearer ${token}`,
      },
    },
  );

  if (!response.ok) {
    let message = getFriendlyError(response.status);

    try {
      const errorData = await response.json();

      // Don't expose raw server exceptions to the UI.
      if (response.status >= 500) {
        message =
          "The server couldn't retrieve this document. Please try again.";
      } else if (response.status === 401 || response.status === 403) {
        message = getFriendlyError(response.status);
      } else if (errorData.message) {
        message = errorData.message;
      }
    } catch {
      // Keep the friendly fallback message.
    }

    throw new Error(message);
  }

  return await response.blob();
}

export function downloadBlob(blob, filename) {
  const objectUrl = URL.createObjectURL(blob);

  const link = document.createElement("a");
  link.href = objectUrl;
  link.download = filename || "document";
  document.body.appendChild(link);
  link.click();
  link.remove();

  URL.revokeObjectURL(objectUrl);
}

  let response;

      method: "GET",
      headers: {
        Authorization: `Bearer ${token}`,
        Accept: "*/*",
      },
      cache: "no-store",
    });
  } catch {
    throw new Error(

    );

  if (!response.ok) {
    const fallback =
      action === "preview"
        ? "The server couldn't preview this document. Please try again later."
        : "The server couldn't download this document. Please try again later.";

    const message = await getApiErrorMessage(response, fallback, endpoint);

    throw new Error(message);
  }

  return response.blob();
}

export function formatFileSize(bytes) {
  if (bytes === null || bytes === undefined) {
    return "N/A";
  }

  const size = Number(bytes);

  if (!Number.isFinite(size) || size < 0) {
    return "N/A";
  }

  if (size === 0) {
    return "0 Bytes";
  }

  const units = ["Bytes", "KB", "MB", "GB", "TB"];

  const index = Math.min(
    Math.floor(Math.log(size) / Math.log(1024)),
    units.length - 1,
  );

  return `${(size / Math.pow(1024, index)).toFixed(2)} ${units[index]}`;
}

export function isTextDocument(contentType = "", filename = "") {
  const type = getNormalizedContentType(contentType);

  const extension = String(filename).toLowerCase().split(".").pop();

  const textExtensions = ["txt", "csv", "json", "xml", "log", "md"];

  return (
    type.startsWith("text/") ||
    ["application/json", "application/xml"].includes(type) ||
    textExtensions.includes(extension)
  );
}

export function isPreviewable(contentType = "", filename = "") {
  const type = getNormalizedContentType(contentType);

  const extension = String(filename).toLowerCase().split(".").pop();

  const supportedTypes = [
    "application/pdf",
    "image/png",
    "image/jpeg",
    "image/webp",
    "image/gif",
  ];

  const supportedExtensions = [
    "pdf",
    "png",
    "jpg",
    "jpeg",
    "webp",
    "gif",
    "txt",
    "csv",
    "json",
    "xml",
    "log",
    "md",
  ];

  return (
    supportedTypes.includes(type) ||
    type.startsWith("image/") ||
    isTextDocument(type, filename) ||
    supportedExtensions.includes(extension)
  );
}
